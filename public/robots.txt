User-agent: *
Allow: /

# L'administration n'a rien à faire dans un index : elle exige une session, et ses URL
# indexées ne produiraient que des redirections vers l'écran de connexion.
Disallow: /config/
Disallow: /en/config/
Disallow: /login
Disallow: /en/login

# Même raison pour les écrans de l'outil LoL et le profil : ils exigent une session. La vitrine
# publique, elle, est `/lol` — et elle doit être indexée.
Disallow: /lol/teams
Disallow: /en/lol/teams
Disallow: /lol/stats
Disallow: /en/lol/stats
Disallow: /profile
Disallow: /en/profile
